"""Core data types: stages, data kinds, actions, updates, findings, decisions."""
from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum

import torch


class Stage(str, Enum):
    """Where in a federated round an object crosses a trust boundary, in the order a round reaches them.

    Not tied to any attack or defense. A plugin may mount at several stages of its side.
    """
    SERVER_DISTRIBUTE = "server_distribute"          # the global model about to be sent, per recipient
    CLIENT_ANNOUNCE = "client_announce"              # model and data are in; the client may send a summary first
    SERVER_EXCHANGE = "server_exchange"              # the server answers those summaries, before any training
    CLIENT_PRE_TRAIN = "client_pre_train"            # local data and the received global model
    CLIENT_LOSS = "client_loss"                      # the batch loss is computed, backward has not run (trainer extension only)
    CLIENT_PRE_STEP = "client_pre_step"              # gradients are computed, the optimizer has not stepped
    CLIENT_STEP = "client_step"                      # after each local training step
    CLIENT_POST_TRAIN = "client_post_train"          # local result, before upload
    SERVER_RECEIVE = "server_receive"                # uploads arrive
    SERVER_AGGREGATE = "server_aggregate"            # who is aggregated, with what weight
    SERVER_POST_AGGREGATE = "server_post_aggregate"  # the new global model, every round


# Data kinds a plugin may ask for, and the side that handles them. A side only ever sees what
# passes through it; neither side is trusted by default.
KIND_SIDE: dict[str, str] = {
    "data": "client",            # local samples and the data manifest
    "received_model": "client",  # the global LoRA this client received
    "train_trace": "client",     # per-step scalars, and the current LoRA on request
    "loss": "client",            # this step's batch, model outputs and loss, before backward
    "gradients": "client",       # this step's gradients of the trainable parameters
    "local_update": "client",    # the local LoRA before upload
    "outgoing": "server",        # the global LoRA about to be sent, and to whom
    "uploads": "server",         # per-client uploaded LoRA
    "aggregate": "server",       # participants, weights, aggregation result
    "reports": "both",           # self-reported values and channel messages; never verified
    "model_handle": "both",      # a model the plugin may run forward/backward passes on
    "history": "both",           # this side's records of earlier rounds
}
KINDS = frozenset(KIND_SIDE)

_CLIENT = (Stage.CLIENT_ANNOUNCE, Stage.CLIENT_PRE_TRAIN, Stage.CLIENT_LOSS, Stage.CLIENT_PRE_STEP, Stage.CLIENT_STEP, Stage.CLIENT_POST_TRAIN)
_SERVER = (Stage.SERVER_DISTRIBUTE, Stage.SERVER_EXCHANGE, Stage.SERVER_RECEIVE, Stage.SERVER_AGGREGATE,
           Stage.SERVER_POST_AGGREGATE)
STAGE_SIDE: dict[Stage, str] = {**{s: "client" for s in _CLIENT}, **{s: "server" for s in _SERVER}}

_COMMON = {"reports", "model_handle", "history"}
_LOCAL = _COMMON | {"data", "received_model"}
STAGE_KINDS: dict[Stage, frozenset[str]] = {
    Stage.SERVER_DISTRIBUTE: frozenset(_COMMON | {"outgoing"}),
    Stage.CLIENT_ANNOUNCE: frozenset(_LOCAL),
    Stage.SERVER_EXCHANGE: frozenset(_COMMON),
    Stage.CLIENT_PRE_TRAIN: frozenset(_LOCAL),
    Stage.CLIENT_LOSS: frozenset(_LOCAL | {"train_trace", "loss"}),
    Stage.CLIENT_PRE_STEP: frozenset(_LOCAL | {"train_trace", "gradients"}),
    Stage.CLIENT_STEP: frozenset(_LOCAL | {"train_trace"}),
    Stage.CLIENT_POST_TRAIN: frozenset(_LOCAL | {"train_trace", "local_update"}),
    Stage.SERVER_RECEIVE: frozenset(_COMMON | {"uploads"}),
    Stage.SERVER_AGGREGATE: frozenset(_COMMON | {"uploads", "aggregate"}),
    Stage.SERVER_POST_AGGREGATE: frozenset(_COMMON | {"uploads", "aggregate"}),
}

# Interventions. Off unless the runner is told to apply them (for all plugins, or by name).
ACTION_STAGES: dict[str, frozenset[Stage]] = {
    "transform_broadcast": frozenset({Stage.SERVER_DISTRIBUTE}),
    "transform_received": frozenset({Stage.CLIENT_PRE_TRAIN}),
    "mask_samples": frozenset({Stage.CLIENT_PRE_TRAIN}),
    "edit_samples": frozenset({Stage.CLIENT_PRE_TRAIN}),
    "transform_loss": frozenset({Stage.CLIENT_LOSS}),
    "transform_gradients": frozenset({Stage.CLIENT_PRE_STEP}),
    "transform_update": frozenset({Stage.CLIENT_POST_TRAIN, Stage.SERVER_RECEIVE}),
    "select_and_weight": frozenset({Stage.SERVER_AGGREGATE}),
    "replace_global": frozenset({Stage.SERVER_POST_AGGREGATE}),
}


def side_of(stage: Stage) -> str:
    return STAGE_SIDE[stage]


@dataclass(frozen=True)

    """Static facts about a run that plugins may rely on.

    `unavailable` lists data kinds this deployment cannot provide (e.g. {"uploads"} under secure
    aggregation); plugins needing them are refused before the first round.
    """
    run_id: str
    seed: int = 0
    lora_scale: float = 1.0          # alpha / r
    num_clients: int | None = None
    unavailable: frozenset[str] = frozenset()
    extra: dict = field(default_factory=dict)


@dataclass
class ClientUpdate:
    """One client's upload in one round, as seen by the server.

    `lora` and `base` are PEFT state dicts (``...lora_A.weight`` / ``...lora_B.weight``).
    `reported` holds values the client claims (e.g. num_examples, train_loss); they are
    never verified by construction and must be treated as untrusted.
    """
    client_id: str
    round: int
    lora: dict[str, torch.Tensor]
    base: dict[str, torch.Tensor]
    reported: dict[str, float] = field(default_factory=dict)
    messages: dict[str, object] = field(default_factory=dict)   # channel messages from the client's plugins; untrusted
    attachments: dict[str, torch.Tensor] = field(default_factory=dict)   # tensors sent over the channel; untrusted


MESSAGE_LIMIT_BYTES = 65536


def check_messages(messages: dict) -> None:
    """Channel messages are small typed records: numbers, strings, booleans, or flat lists of them."""
    import json
    for k, v in messages.items():
        items = v if isinstance(v, list) else [v]
        if not isinstance(k, str) or not k or not all(isinstance(x, (int, float, str, bool)) for x in items):
            raise TypeError(f"message {k!r}: only numbers, strings, booleans and flat lists of them can be sent")
    if len(json.dumps(messages)) > MESSAGE_LIMIT_BYTES:
        raise ValueError(f"messages exceed {MESSAGE_LIMIT_BYTES} bytes; the channel is for small records, not tensors")


@dataclass
class Decision:
    """The `select_and_weight` action proposed by a plugin at SERVER_AGGREGATE.

    weights: absolute aggregation weights per client (clients not listed keep their weight).
    excluded: clients removed from this round's aggregation.
    flags: human-readable reasons per client.
    """
    weights: dict[str, float] = field(default_factory=dict)
    excluded: set[str] = field(default_factory=set)
    flags: dict[str, str] = field(default_factory=dict)

    @staticmethod
    def merge(decisions: list["Decision"]) -> "Decision":
        """Union of exclusions, element-wise minimum of weights.

        Stacking several detectors this way can only remove more clients, never fewer; with
        correlated detectors it amplifies false exclusions.
        """
        out = Decision()
        for d in decisions:
            out.excluded |= d.excluded
            for cid, w in d.weights.items():
                out.weights[cid] = min(w, out.weights.get(cid, w))
            for cid, why in d.flags.items():
                out.flags[cid] = f"{out.flags[cid]}; {why}" if cid in out.flags else why
        return out


@dataclass
class Findings:
    """Output of one plugin at one stage in one round.

    The plugin's name is stored under `module` so that records written before the stage/kind
    contract stay readable.
    """
    module: str
    stage: Stage
    round: int
    scores: dict[str, float] = field(default_factory=dict)             # subject -> risk score
    features: dict[str, dict[str, float]] = field(default_factory=dict)  # subject -> name -> value
    evidence: dict = field(default_factory=dict)
    decision: Decision | None = None
    messages: dict[str, object] = field(default_factory=dict)          # sent to the other side as '<plugin>.<key>'
    # Tensors sent to the other side as '<plugin>.<key>'. The runner moves them to its outbox and leaves
    # their hashes in `sent`; nothing here is written as numbers.
    attachments: dict = field(default_factory=dict)
    sent: dict[str, str] = field(default_factory=dict)
    # Interventions other than `decision`, as the plugin proposes them. The runner turns them into
    # `actions` records (hashes, not tensors) and empties this field before anything is written.
    #   "broadcast": {client id or "*": LoRA state dict}   (transform_broadcast)
    #   "received": LoRA state dict                        (transform_received)
    #   "mask": iterable of sample hashes                  (mask_samples)
    #   "add": list of samples, "rewrite": {hash: sample}  (edit_samples)
    #   "loss": scalar tensor with a graph                 (transform_loss)
    #   "gradients": {parameter name: tensor}              (transform_gradients)
    #   "transformed": {subject: LoRA state dict}          (transform_update)
    #   "replacement": LoRA state dict                     (replace_global)
    proposals: dict = field(default_factory=dict)
    actions: list[dict] = field(default_factory=list)
    cost: dict[str, float] = field(default_factory=dict)   # model passes the plugin spent, as the handle counted them
    error: str | None = None
    elapsed_s: float = 0.0

===== fedguard/core/plugin.py  (import as fedguard.core.plugin) =====
| Base class for plugins and the checks run on their declarations.
class ContractError(ValueError):
    | A plugin's declaration cannot be satisfied; raised before the first round.

class GuardPlugin:
    | An algorithm mounted at one or several stages of one side.
    | 
    | A plugin declares four things:
    |   at         the stage it runs at, or a tuple of stages; `run` is called once per stage and can
    |              branch on `ctx.stage`. Its private state is shared between its stages, in three scopes:
    |              `ctx.state` for the whole run, `ctx.round_state` for one round, `ctx.step_state` for
    |              one local training step (the runner empties the last two when their scope ends).
    |   needs      the data kinds it reads (see fedguard.core.types.KIND_SIDE); anything else is refused
    |   resources  names of defender-supplied assets it uses (clean reference data, probe sets, ...);
    |              the caller passes them to the runner, the plugin reads them with ctx.resource(name)
    |   acts       the interventions it may propose; empty means report-only
    | 
    | Subclasses compute from `ctx` views and `fedguard.ops` operators and return `ctx.findings(...)`.
    name: str = ''
    at: Stage | tuple[Stage, ...]
    needs: frozenset[str] = frozenset()
    resources: frozenset[str] = frozenset()
    acts: frozenset[str] = frozenset()
    version: str = '0'
    @property def stages(self) -> tuple[Stage, ...]
    def setup(self, spec: RunSpec) -> None
        | Called once before the first round.
    def run(self, ctx) -> Findings

def check_contract(plugin: GuardPlugin, side: str, spec: RunSpec, resources) -> None
    | Refuse a plugin whose declaration cannot hold at its stages, on this side, in this deployment.

===== fedguard/client.py  (import as fedguard.client) =====
| Client-side capture: what one client received, trained on, did, and is about to upload.
| 
| Framework-neutral. A training loop (or an adapter: fedguard.adapters.flower_client,
| fedguard.adapters.hf_trainer) calls, per round:
| 
|     guard.begin_round(round, received_lora, data=DataView(samples, text_of=...))   # CLIENT_PRE_TRAIN
|     samples = guard.training_samples()                                             # after applied sample actions
|     loss = guard.on_loss(step, LossView(...))                                      # CLIENT_LOSS, trainer extension only
|     guard.on_gradients(step, model.named_parameters())                             # CLIENT_PRE_STEP, before optimizer.step()
|     guard.on_step(step, {"loss": ..., "lr": ...}, lora=lambda: lora_state(model))  # CLIENT_STEP, after each step
|     guard.snapshot(step, lora)                                                     # optional, every snapshot_every steps
|     guard.before_upload(local_lora, {"num_examples": n})                           # CLIENT_POST_TRAIN
|     guard.end_round()
| 
| When client plugins need an answer from the server before training starts, the round opens with an
| exchange instead: begin_round(..., defer=True); announce() runs CLIENT_ANNOUNCE and returns what to send;
| deliver(...) hands in the server's reply; pre_train() then runs CLIENT_PRE_TRAIN.
| 
| The client's record mirrors the server's (see fedguard.recorder.store): `received` and `local` are
| content-hashed, so they can be compared with the server's `base` and `client_<id>` after the fact.
| Sample text is never written; the manifest holds hashes, source tags and lengths.
class ClientGuard:
    | One client's side of the library for the whole run: holds what the client received and is training on,
    | runs the client-side stages when the training code (or an adapter) reports that a point was reached,
    | and writes the client's own record. One instance per client; `begin_round` ... `end_round` per round.
    def __init__(self, client_id: str, runner: GuardRunner | None=None, recorder=None, snapshot_every: int=0)
        | runner: a GuardRunner(side="client"); None captures without running plugins.
        | recorder: a Recorder(scope=client_id); None runs plugins without recording.
        | snapshot_every: store the LoRA factors handed to snapshot() every N steps (0 = never).
    def begin_round(self, round_idx: int, received: dict[str, torch.Tensor], data: DataView | None=None, model_handle=None, messages: dict | None=None, defer: bool=False, attachments: dict | None=None) -> list[Findings]
        | Record what arrived and run CLIENT_PRE_TRAIN. `messages` and `attachments` are what the
        | server's plugins sent over the channel.
        | 
        | defer=True only records the received model; the stage runs when pre_train() is called. This is
        | for adapters that see the message before the training code has loaded its data, and for rounds
        | that open with an announce exchange.
    @property def wants_exchange(self) -> bool
        | True when a mounted plugin runs at CLIENT_ANNOUNCE, i.e. the round should open with an exchange.
    @property def announced_round(self) -> int | None
        | The round that was opened with an announce and has not been trained yet; None otherwise.
    def announce(self, data: DataView | None=None, model_handle=None) -> tuple[dict, dict]
        | Run CLIENT_ANNOUNCE and return (messages, tensors) for the server's SERVER_EXCHANGE stage.
    def deliver(self, messages: dict | None=None, attachments: dict | None=None) -> None
        | Hand in what the server answered during the exchange; visible to CLIENT_PRE_TRAIN and later stages.
    def pre_train(self, data: DataView | None=None, model_handle=None, timing: str='on_time') -> list[Findings]
        | Record the data manifest and run CLIENT_PRE_TRAIN. Once per round.
    def start_state(self) -> dict[str, torch.Tensor]
        | The LoRA to start local training from: what was received, or what an applied
        | `transform_received` made of it. The upload is still measured against what was received.
    def training_samples(self) -> list
        | The samples to train on this round: the announced data after every applied `mask_samples` and
        | `edit_samples` action. Without such actions this is the announced data, in order.
    def training_hashes(self) -> list[str]
        | The hash of each of training_samples(), in the same order: the manifest hash for an untouched
        | sample, the hash of the new text for a rewritten or added one.
    def log(self, step: int, scalars: dict[str, float]) -> None
        | Record a step's scalars without running plugins. Scalars of the same step are merged.
    def on_loss(self, step: int, view: LossView) -> torch.Tensor
        | The batch loss is computed and backward has not run: run CLIENT_LOSS and return the loss to
        | back-propagate, which is the host's own tensor unless an applied `transform_loss` replaced it.
    def on_gradients(self, step: int, named_parameters) -> None
        | Gradients are computed and the optimizer has not stepped: run CLIENT_PRE_STEP.
        | Does nothing unless a plugin is mounted there.
    def on_step(self, step: int, scalars: dict[str, float], lora=None) -> None
        | Record one finished training step and run CLIENT_STEP. `lora` is a function returning the
        | current LoRA state; it is only called if a plugin asks for it. Without plugins at this stage
        | the cost is a dict update.
    def wants_snapshot(self, step: int) -> bool
    def snapshot(self, step: int, lora: dict[str, torch.Tensor]) -> None
        | Store the LoRA factors at this step if it falls on the snapshot interval.
    def before_upload(self, local: dict[str, torch.Tensor], reported: dict[str, float] | None=None) -> list[Findings]
        | Record the local result and what the client is about to claim, then run CLIENT_POST_TRAIN.
    def kept_indices(self) -> list[int]
        | Indices of this round's samples that no applied `mask_samples` action removed. Train on these.
    def upload_state(self) -> dict[str, torch.Tensor]
        | The LoRA to upload: the local result, or what an applied `transform_update` made of it.
    def outbox(self) -> dict[str, object]
        | Messages this round's client plugins addressed to the server; the adapter sends them with the upload.
    def outbox_tensors(self) -> dict[str, torch.Tensor]
        | Tensors this round's client plugins addressed to the server.
    def end_round(self, meta: dict | None=None) -> None

| Clustering of a small number of points (clients, or a client's samples). Torch only.

    | Lloyd's algorithm with k-means++ seeding; the best of `restarts` runs by inertia. Returns (labels, centers).

    | Bottom-up clustering from a distance matrix until `n_clusters` remain. linkage: average | single | complete.

def dbscan(D: torch.Tensor, eps: float, min_samples: int) -> torch.Tensor
    | DBSCAN on a distance matrix. A point is a core point when at least `min_samples` points (itself
    | included) lie within eps. Noise is labelled -1.

def majority_cluster(D: torch.Tensor, min_size: int | None=None, gap: float=2.0) -> torch.Tensor
    | The dense group that holds a majority of the points; returns a boolean membership mask.
    | 
    | FLAME and FreqFed run HDBSCAN with min_cluster_size = n // 2 + 1 and keep that single cluster.
    | When scikit-learn is installed this calls its HDBSCAN with exactly that setting. Otherwise it uses a
    | substitute with the same contract, not the same algorithm:
    |   1. merge points by single linkage in order of increasing distance until one component holds
    |      `min_size` points; call the distance of that merge the core level;
    |   2. keep attaching the nearest outside point, noting the distance at which each one joins;
    |   3. cut at the first place where the joining distance jumps by a factor of `gap` or more over the
    |      previous level. Points beyond the cut are outliers; with no such jump, every point is kept.
    | If no majority forms, all points are kept.

def density_clusters(D: torch.Tensor, min_samples: int=3, scale: float=2.0) -> torch.Tensor
    | DBSCAN with the radius read off the data: eps = scale * median distance to the (min_samples - 1)-th
    | nearest neighbour. Finds however many dense groups there are; sparse points are noise (-1).
    | A stand-in where a method asks for HDBSCAN without giving its parameters.

def silhouette(D: torch.Tensor, labels: torch.Tensor) -> float
    | Mean silhouette coefficient from a distance matrix; noise points (-1) are ignored. Needs >= 2 clusters.

def distances(X: torch.Tensor, metric: str='euclidean') -> torch.Tensor
    | Distance matrix between rows: euclidean, or cosine (1 - cosine similarity).

def gap_statistic_k(x: torch.Tensor, k_max: int=10, references: int=20, seed: int=0) -> int
    | Number of clusters in a 1-D sample by the gap statistic (Tibshirani et al. 2001), as FLDetector
    | uses it: min-max scale, compare log within-cluster dispersion against uniform reference samples, and
    | return the smallest k with Gap(k) >= Gap(k+1) - s(k+1). Returns 1 when nothing beats one cluster.

===== fedguard/ops/lowrank.py  (import as fedguard.ops.lowrank) =====
| Effect-space primitives for LoRA: exact norms, inner products and spectra without densifying.
| 
| A matrix is represented as M = scale * U @ V with U: (d_out, q), V: (q, d_in), q small.
| PEFT stores lora_B as (d_out, r) and lora_A as (r, d_in), so a LoRA effect is (U, V) = (B, A).
| 
| A round's functional change is rank <= 2r:
|     B_i A_i - B_g A_g = [B_i, -B_g] @ [A_i; A_g]
| Do NOT use (B_i - B_g)(A_i - A_g): it drops the first-order terms that make up most of the update.
class LowRank:
    | A matrix kept as scale * U @ V without forming it. Norms, inner products and singular values of
    | LoRA products and of their differences are computed from the small Gram matrices of the factors.
    U: torch.Tensor
    V: torch.Tensor
    scale: float = 1.0
    @classmethod def lora(cls, B: torch.Tensor, A: torch.Tensor, scale: float=1.0) -> 'LowRank'
        | Cumulative LoRA effect scale * B @ A.
    @classmethod def increment(cls, B: torch.Tensor, A: torch.Tensor, B0: torch.Tensor, A0: torch.Tensor, scale: float=1.0) -> 'LowRank'
        | Functional change scale * (B @ A - B0 @ A0) relative to the broadcast factors (B0, A0).
    @property def shape(self) -> tuple[int, int]
    def to(self, dtype: torch.dtype) -> 'LowRank'
    def dense(self) -> torch.Tensor
        | Materialize the full matrix. For tests and small models only.
    def norm_sq(self) -> float
    def inner(self, other: 'LowRank') -> float
    def singular_values(self) -> torch.Tensor
        | Nonzero spectrum via QR of each factor and SVD of a q x q core.

def gram(items: list[LowRank], dtype: torch.dtype=torch.float64) -> torch.Tensor
    | Pairwise Frobenius inner products of same-shape low-rank matrices, batched into two GEMMs.
    | 
    | Returns an (n, n) tensor G with G[i, j] = <M_i, M_j>.

def kappa(B: torch.Tensor, A: torch.Tensor) -> float
    | Factorization redundancy ||B||·||A|| / ||BA|| (>= 1). Undefined (inf) when BA = 0.
